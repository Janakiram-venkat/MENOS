// Scratch, Phase 16 step 2 (docs/adr/0020): the specialised kernels' FP64
// instruction rate as a fraction of the device's FP64 peak, measured
// back to back so both see the same GPU clock. The 1q/2q kernels are verbatim
// copies of src/kernels/cuda/dense_kernels.cu at f495da6. Build:
//   nvcc -O3 -fmad=false -arch=sm_86 -x cu fp64_fraction.cu.txt -o fp64_fraction
#include <cuComplex.h>
#include <cuda_runtime.h>

#include <cstdint>
#include <cstdio>

__device__ inline cuDoubleComplex cmul(cuDoubleComplex a, cuDoubleComplex b) {
    return make_cuDoubleComplex(a.x * b.x - a.y * b.y, a.x * b.y + a.y * b.x);
}
__device__ inline cuDoubleComplex cadd(cuDoubleComplex a, cuDoubleComplex b) {
    return make_cuDoubleComplex(a.x + b.x, a.y + b.y);
}
struct Matrix2 { cuDoubleComplex m[4]; };
struct Matrix4 { cuDoubleComplex m[16]; };
__device__ inline std::size_t insert_zero_bit(std::size_t i, std::uint32_t pos) {
    const std::size_t low = (std::size_t{1} << pos) - 1;
    return ((i & ~low) << 1) | (i & low);
}
__global__ void k1(cuDoubleComplex* __restrict__ state, std::size_t n_pairs, std::uint32_t q, Matrix2 matrix) {
    const std::size_t i = static_cast<std::size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    if (i >= n_pairs) return;
    const std::size_t i0 = insert_zero_bit(i, q);
    const std::size_t i1 = i0 | (std::size_t{1} << q);
    const cuDoubleComplex a0 = state[i0];
    const cuDoubleComplex a1 = state[i1];
    cuDoubleComplex out0 = make_cuDoubleComplex(0.0, 0.0);
    out0 = cadd(out0, cmul(matrix.m[0], a0));
    out0 = cadd(out0, cmul(matrix.m[1], a1));
    cuDoubleComplex out1 = make_cuDoubleComplex(0.0, 0.0);
    out1 = cadd(out1, cmul(matrix.m[2], a0));
    out1 = cadd(out1, cmul(matrix.m[3], a1));
    state[i0] = out0;
    state[i1] = out1;
}
__global__ void k2(cuDoubleComplex* __restrict__ state, std::size_t n_groups, std::uint32_t q0, std::uint32_t q1, Matrix4 matrix) {
    const std::size_t i = static_cast<std::size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    if (i >= n_groups) return;
    const std::uint32_t lo = q0 < q1 ? q0 : q1;
    const std::uint32_t hi = q0 < q1 ? q1 : q0;
    const std::size_t b0 = std::size_t{1} << q0;
    const std::size_t b1 = std::size_t{1} << q1;
    const std::size_t base = insert_zero_bit(insert_zero_bit(i, lo), hi);
    const std::size_t idx[4] = {base, base | b0, base | b1, base | b0 | b1};
    const cuDoubleComplex a[4] = {state[idx[0]], state[idx[1]], state[idx[2]], state[idx[3]]};
#pragma unroll
    for (int out = 0; out < 4; ++out) {
        cuDoubleComplex sum = make_cuDoubleComplex(0.0, 0.0);
#pragma unroll
        for (int in = 0; in < 4; ++in) sum = cadd(sum, cmul(matrix.m[out * 4 + in], a[in]));
        state[idx[out]] = sum;
    }
}
// 8 independent DMUL/DADD chains per thread: the FP64 issue ceiling.
__global__ void peak(double* out, double a, double b, int iters) {
    double x[8];
    for (int j = 0; j < 8; ++j) x[j] = threadIdx.x + j;
    for (int i = 0; i < iters; ++i) {
#pragma unroll
        for (int j = 0; j < 8; ++j) { x[j] = x[j] * a; x[j] = x[j] + b; }
    }
    double s = 0; for (int j = 0; j < 8; ++j) s += x[j];
    if (s == 12345.678) out[0] = s;
}
// Reads and writes every amplitude once: the memory floor.
__global__ void touch(cuDoubleComplex* __restrict__ state, std::size_t n, double one) {
    const std::size_t i = static_cast<std::size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    if (i >= n) return;
    cuDoubleComplex v = state[i];
    state[i] = make_cuDoubleComplex(v.x * one, v.y * one);
}

template <typename F> float time_ms(F f) {
    cudaEvent_t e0, e1; cudaEventCreate(&e0); cudaEventCreate(&e1);
    cudaEventRecord(e0); f(); cudaEventRecord(e1); cudaEventSynchronize(e1);
    float ms; cudaEventElapsedTime(&ms, e0, e1);
    cudaEventDestroy(e0); cudaEventDestroy(e1);
    return ms;
}

int main() {
    const int n = 26;
    const std::size_t amps = std::size_t{1} << n;
    cuDoubleComplex* state; cudaMalloc(&state, amps * sizeof(cuDoubleComplex));
    double* out; cudaMalloc(&out, 8);
    cudaMemset(state, 0, amps * sizeof(cuDoubleComplex));
    Matrix2 m2; for (int e = 0; e < 4; ++e) m2.m[e] = make_cuDoubleComplex(0.5, 0.25);
    Matrix4 m4; for (int e = 0; e < 16; ++e) m4.m[e] = make_cuDoubleComplex(0.5, 0.25);
    const int B = 256;
    const int pb = 24 * 64, piters = 1000;
    const double peak_ops = double(pb) * B * piters * 16;
    const double ops1 = double(amps / 2) * 32, ops2 = double(amps / 4) * 128;
    printf("n=%d, rates in G FP64 instr/s, each row measured back to back\n", n);
    printf("%4s %8s %8s %8s %8s %8s %10s %10s\n", "rep", "peak", "1q", "1q/peak", "2q", "2q/peak", "1q ms", "touch ms");
    for (int rep = 0; rep < 8; ++rep) {
        const float tp = time_ms([&] { peak<<<pb, B>>>(out, 0.999999, 1e-9, piters); });
        const float t1 = time_ms([&] { k1<<<(unsigned)((amps / 2 + B - 1) / B), B>>>(state, amps / 2, 13, m2); });
        const float t2 = time_ms([&] { k2<<<(unsigned)((amps / 4 + B - 1) / B), B>>>(state, amps / 4, 13, 14, m4); });
        const float tt = time_ms([&] { touch<<<(unsigned)((amps + B - 1) / B), B>>>(state, amps, 1.0); });
        const double rp = peak_ops / (tp * 1e6), r1 = ops1 / (t1 * 1e6), r2 = ops2 / (t2 * 1e6);
        printf("%4d %8.1f %8.1f %8.2f %8.1f %8.2f %10.2f %10.2f\n", rep, rp, r1, r1 / rp, r2, r2 / rp, t1, tt);
    }
    return 0;
}
